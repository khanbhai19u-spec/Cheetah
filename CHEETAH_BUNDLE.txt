Future> tmdb(String path, [Map<String, String> q = const {}]) async {
  final uri = Uri.https('api.themoviedb.org', '/3$path',
      {'api_key': tmdbKey, 'language': 'en-US', ...q});
  final r = await http.get(uri);
  if (r.statusCode != 200) {
    throw Exception('TMDB error ${r.statusCode}');
  }
  return json.decode(r.body) as Map<String, dynamic>;
}

class Movie {
  final int id;
  final String title;
  final String? poster;
  final String? backdrop;
  final double rating;
  final String year;
  final String overview;

  Movie({
    required this.id,
    required this.title,
    required this.poster,
    required this.backdrop,
    required this.rating,
    required this.year,
    required this.overview,
  });

  factory Movie.fromJson(Map<String, dynamic> j) {
    final date = (j['release_date'] ?? '') as String;
    return Movie(
      id: j['id'] as int,
      title: (j['title'] ?? j['name'] ?? '') as String,
      poster: j['poster_path'] as String?,
      backdrop: j['backdrop_path'] as String?,
      rating: ((j['vote_average'] ?? 0) as num).toDouble(),
      year: date.length >= 4 ? date.substring(0, 4) : '',
      overview: (j['overview'] ?? '') as String,
    );
  }
}

Future<List<Movie>> fetchMovies(String path, [Map<String, String> q = const {}]) async {
  final d = await tmdb(path, q);
  return (d['results'] as List)
      .map((e) => Movie.fromJson(e as Map<String, dynamic>))
      .where((m) => m.poster != null)
      .toList();
}

Future<String> getDirectVideoUrl(String rawUrl) async {
  if (rawUrl.isEmpty) return rawUrl;
  final apiUri = Uri.parse('https://cheetah-bd.vercel.app/api/get-link?url=${Uri.encodeComponent(rawUrl)}');
  try {
    final res = await http.get(apiUri).timeout(const Duration(seconds: 10));
    if (res.statusCode == 200) {
      final data = jsonDecode(res.body);
      if (data['success'] == true && data['direct_url'] != null) {
        return data['direct_url'];
      }
    }
  } catch (_) {}
  return rawUrl;
}

    _debounce?.cancel();
    _debounce = Timer(const Duration(milliseconds: 500), () => _search(v));
  }

  Future<void> _search(String v) async {
    final q = v.trim();
    if (q.isEmpty) {
      setState(() {
        _results = [];
        _query = '';
      });
      return;
    }
    setState(() {
      _loading = true;
      _query = q;
    });
    try {
      final r = await fetchMovies('/search/movie', {'query': q});
      if (!mounted) return;
      setState(() => _results = r);
    } catch (_) {
      if (!mounted) return;
      setState(() => _results = []);
    }
    if (mounted) setState(() => _loading = false);
  }

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      child: Column(children: [
        Padding(
          padding: const EdgeInsets.all(16),

enum DlStatus { downloading, paused, done, failed }

class DownloadItem {
  final int id;
  final String title;
  final String? poster;
  final String url;
  final String fileName;
  int received;
  int total;
  DlStatus status;

  DownloadItem({
    required this.id,
    required this.title,
    required this.poster,
    required this.url,
    required this.fileName,
    this.received = 0,
    this.total = 0,
    this.status = DlStatus.paused,
  });

  double get progress =>
      total > 0 ? (received / total).clamp(0.0, 1.0).toDouble() : 0.0;

  Map<String, dynamic> toJson() => {
        'id': id,
        'title': title,
        'poster': poster,
        'url': url,
        'fileName': fileName,
        'received': received,
        'total': total,
        'status': status.index,
      };

  factory DownloadItem.fromJson(Map<String, dynamic> j) => DownloadItem(
        id: j['id'] as int,
        title: j['title'] as String,
        poster: j['poster'] as String?,
        url: j['url'] as String,
        fileName: j['fileName'] as String,
        received: (j['received'] ?? 0) as int,
        total: (j['total'] ?? 0) as int,
        status: DlStatus.values[(j['status'] ?? 1) as int],
      );
}

String fmtBytes(int b) {
  if (b < 1024) return '$b B';
  if (b < 1024 * 1024) return '${(b / 1024).toStringAsFixed(0)} KB';
  if (b < 1024 * 1024 * 1024) {
    return '${(b / 1024 / 1024).toStringAsFixed(1)} MB';
  }
  return '${(b / 1024 / 1024 / 1024).toStringAsFixed(2)} GB';
}

class DownloadManager extends ChangeNotifier {
  DownloadManager._();
  static final DownloadManager I = DownloadManager._();

  final Dio _dio = Dio(BaseOptions(connectTimeout: const Duration(seconds: 20)));
  final Map<int, DownloadItem> items = {};
  final Map<int, CancelToken> _tokens = {};
  late Directory _dir;
  SharedPreferences? _prefs;
  DateTime _lastNotify = DateTime.fromMillisecondsSinceEpoch(0);

  Future<void> init() async {
    final base = await getApplicationDocumentsDirectory();
    _dir = Directory('${base.path}/downloads');
    if (!await _dir.exists()) await _dir.create(recursive: true);
    _prefs = await SharedPreferences.getInstance();
    final raw = _prefs!.getString('downloads');
    if (raw != null) {
      for (final e in json.decode(raw) as List) {
        final it = DownloadItem.fromJson(e as Map<String, dynamic>);
        if (it.status == DlStatus.downloading) it.status = DlStatus.paused;
        items[it.id] = it;
      }
    }
  }

  List<DownloadItem> get list => items.values.toList().reversed.toList();

  DownloadItem? get(int id) => items[id];

  String pathOf(DownloadItem it) => '${_dir.path}/${it.fileName}';

  String? localPath(int id) {
    final it = items[id];
    if (it == null || it.status != DlStatus.done) return null;
    final p = pathOf(it);
    return File(p).existsSync() ? p : null;
  }

  Future<void> _save() async {
    await _prefs?.setString(
        'downloads', json.encode(items.values.map((e) => e.toJson()).toList()));
  }

  void start(int id, String title, String? poster, String url) {
    final existing = items[id];
    if (existing != null) {
      if (existing.status == DlStatus.paused ||
          existing.status == DlStatus.failed) {
        resume(id);
      }
      return;
    }
    final it = DownloadItem(
      id: id,
      title: title,
      poster: poster,
      url: url,
      fileName: 'movie_$id.mp4',
      status: DlStatus.downloading,
    );
    items[id] = it;
    notifyListeners();
    _run(it);
  }

  void pause(int id) => _tokens[id]?.cancel('pause');

  void resume(int id) {
    final it = items[id];
    if (it == null ||
        it.status == DlStatus.downloading ||
        it.status == DlStatus.done) {
      return;
    }
    _run(it);
  }

  Future<void> remove(int id) async {
    final it = items.remove(id);
    _tokens[id]?.cancel('remove');
    _tokens.remove(id);
    if (it != null) {
      try {
        final f = File(pathOf(it));
        if (await f.exists()) await f.delete();
      } catch (_) {}
    }
    await _save();
    notifyListeners();
  }

  Future<void> _run(DownloadItem it) async {
    final file = File(pathOf(it));
    var existing = await file.exists() ? await file.length() : 0;

    if (it.total > 0 && existing >= it.total) {
      it.received = existing;
      it.status = DlStatus.done;
      await _save();
      notifyListeners();
      return;
    }

    final token = CancelToken();
    _tokens[it.id] = token;
    it.status = DlStatus.downloading;
    notifyListeners();

    try {
      final res = await _dio.get<ResponseBody>(
        it.url,
        cancelToken: token,
        options: Options(
          responseType: ResponseType.stream,
          receiveTimeout: Duration.zero,
          headers: existing > 0 ? {'range': 'bytes=$existing-'} : null,
          validateStatus: (s) => s == 200 || s == 206,
        ),
      );
      final partial = res.statusCode == 206;
      if (!partial) existing = 0;
      final len =
          int.tryParse(res.headers.value(Headers.contentLengthHeader) ?? '') ??
              0;
      it.total = len > 0 ? existing + len : 0;
      it.received = existing;

      final sink =
          file.openWrite(mode: partial ? FileMode.append : FileMode.write);
      try {
        await for (final chunk in res.data!.stream) {
          sink.add(chunk);
          it.received += chunk.length;
          final now = DateTime.now();
          if (now.difference(_lastNotify).inMilliseconds > 300) {
            _lastNotify = now;
            notifyListeners();
          }
        }
      } finally {
        await sink.close();
      }
      if (it.total > 0 && it.received < it.total) {
        it.status = DlStatus.paused;
      } else {
        it.status = DlStatus.done;
      }
    } catch (_) {
      it.status = token.isCancelled ? DlStatus.paused : DlStatus.failed;
    }
    _tokens.remove(it.id);
    await _save();
    notifyListeners();
  }
}

import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart' show rootBundle;
import 'package:http/http.dart' as http;
import 'package:package_info_plus/package_info_plus.dart';
import 'package:url_launcher/url_launcher.dart';

import 'core.dart';

Future<String?> _readRepo() async {
  try {
    final t = await rootBundle.loadString('app.config');
    for (final line in t.split('\n')) {
      final l = line.trim();
      if (l.startsWith('GITHUB_REPO=')) {
        return l.substring('GITHUB_REPO='.length).replaceAll('"', '').trim();
      }
    }
  } catch (_) {}
  return null;
}

List<int> _parts(String v) => v
    .replaceAll(RegExp(r'[^0-9.]'), '')
    .split('.')
    .where((e) => e.isNotEmpty)
    .map((e) => int.tryParse(e) ?? 0)
    .toList();

bool _isNewer(String latest, String current) {
  final a = _parts(latest);
  final b = _parts(current);
  for (var i = 0; i < 3; i++) {
    final x = i < a.length ? a[i] : 0;
    final y = i < b.length ? b[i] : 0;
    if (x != y) return x > y;
  }
  return false;
}

Future<void> checkForUpdate(BuildContext context) async {
  try {
    final repo = await _readRepo();
    if (repo == null || repo.isEmpty || repo.contains('YOUR_GITHUB')) return;

    final info = await PackageInfo.fromPlatform();
    final r = await http.get(
      Uri.https('api.github.com', '/repos/$repo/releases/latest'),
      headers: {'Accept': 'application/vnd.github+json'},
    );
    if (r.statusCode != 200) return;

    final j = json.decode(r.body) as Map<String, dynamic>;
    final tag = (j['tag_name'] ?? '') as String;
    if (!_isNewer(tag, info.version)) return;

    String? apkUrl;
    for (final a in (j['assets'] as List? ?? [])) {
      final name = (a['name'] ?? '') as String;
      if (name.endsWith('.apk')) {
        apkUrl = a['browser_download_url'] as String?;
        break;
      }
    }
    if (apkUrl == null || !context.mounted) return;
    final url = apkUrl;

    await showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        backgroundColor: kCard,
        title: const Text('নতুন ভার্সন এসেছে'),
        content: Text('Cheetah $tag পাওয়া যাচ্ছে।\nতোমার ভার্সন: ${info.version}'),
        actions: [
          TextButton(
              onPressed: () => Navigator.pop(ctx),
              child: const Text('পরে')),
          FilledButton(
            style: FilledButton.styleFrom(
                backgroundColor: kOrange, foregroundColor: Colors.black),
            onPressed: () {
              Navigator.pop(ctx);
              launchUrl(Uri.parse(url), mode: LaunchMode.externalApplication);
            },
            child: const Text('আপডেট করো'),
          ),
        ],
      ),
    );
  } catch (_) {}
}
